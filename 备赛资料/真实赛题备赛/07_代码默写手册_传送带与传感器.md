> 这三块代码量少，但现场必用。背会后加在上料机 Main.py 里。

## 传送带电机控制

```python
# 启动传送带（正转，向前送料）
dType.SetEMotor(api, 0, 1, 10000, 1)

# 停传送带
dType.SetEMotor(api, 0, 0, -10000, 1)
```

**参数顺序记忆：api, 端口0, 使能(1开0停), 速度, isQueued=1**
**口诀：端口0，使能1开0停，速度10000**

---

## 光电传感器（检测料是否到位）

```python
# 初始化（程序开头做一次）
dType.SetInfraredSensor(api, 1, 1, version=1)
#                            ↑  ↑
#                         使能  端口1（GP2）

# 循环检测，等料到位
while True:
    i = dType.GetInfraredSensor(api, 1)
    if i[0] == 1:      # 1=检测到料
        break
```

**口诀：先 Set 使能，再 Get 读值，i[0]==1 就是有料**

## 读取当前坐标（示教用，读取座标.py 的核心）

```python
Pose = dType.GetPose(api)
x = round(Pose[0], 4)
y = round(Pose[1], 4)
z = round(Pose[2], 4)
r = round(Pose[3], 4)
print([x, y, z, r])
```

**用法：手动拖臂到目标点 → 运行这几行 → 记录打印出来的坐标 → 填入 Main.py 的 P1/P2/P3 等**

## 上料机完整骨架（把以上拼起来）

```python
import DobotDllType as dType
import socket
import time

api = dType.load()
state = dType.ConnectDobot(api, "COM16", 115200)[0]  # ← 改COM号
print("连接状态:", ["已连接", "未找到设备", "端口占用"][state])

dType.ClearAllAlarmsState(api)
dType.SetPTPJumpParams(api, 100, 110, isQueued=1)
dType.SetEndEffectorParams(api, 59.7, 0, 0, 1)
dType.SetQueuedCmdClear(api)
dType.SetQueuedCmdStartExec(api)
dType.SetEndEffectorGripper(api, 0, 0, isQueued=1)
dType.SetEMotor(api, 0, 0, -10000, 1)            # 传送带初始停止
dType.SetInfraredSensor(api, 1, 1, version=1)    # 使能光电
dType.SetHOMECmd(api, 1, isQueued=1)

# 关键坐标（现场用读取座标.py示教后填入）
P1 = [247.0, 14.8, -70.6, -76.9]   # 码垛抓取起点
P2 = [-102.2, -274.7, 30.4, -169.4]  # RFID位
P3 = [4.2, -270.1, -3.4, -169.4]    # 传送带

x = 0
y = 0

def move(x, y, z, r, air1, air2, keep):
    dType.SetEndEffectorGripper(api, air1, 1, isQueued=1)
    dType.SetPTPCmdEx(api, 1, x, y, z + 50, r, isQueued=1)
    dType.SetPTPCmdEx(api, 1, x, y, z, r, isQueued=1)
    dType.SetEndEffectorGripper(api, air2, 1, isQueued=1)
    dType.dSleep(100)
    dType.SetPTPCmdEx(api, 1, x, y, z + 50, r, isQueued=1)
    if keep == 0:
        dType.SetEndEffectorGripper(api, 0, 0, isQueued=1)

socket_client = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
socket_client.connect(("127.0.0.1", 8081))
print('连接成功')

while True:
    data = socket_client.recv(1024).decode('utf-8')
    print(data)
    if data != 'run':
        continue

    # 码垛算法算当前件的坐标
    P4_y = P1[1] - (y * 50)
    P4_x = P1[0] - (x * 50)
    if y < 3:
        y += 1
    else:
        y = 0
        x = 1

    move(P4_x, P4_y, P1[2], P1[3], 0, 1, 1)           # 抓料
    move(P2[0], P2[1], P2[2], P2[3], 1, 0, 0)           # 放RFID位
    dType.SetPTPCmdEx(api, 1, P3[0], P3[1], P2[2]+60, P2[3], isQueued=1)  # 回避

    socket_client.send("arrive1".encode('utf-8'))

    # 等下料机发OK
    while True:
        data = socket_client.recv(1024).decode('utf-8')
        if 'OK' in data:
            break

    move(P2[0], P2[1], P2[2], P2[3], 0, 1, 1)           # 从RFID抓
    move(P3[0], P3[1], P3[2], P3[3], 1, 0, 0)           # 放传送带

    dType.SetEMotor(api, 0, 1, 10000, 1)                  # 启动传送带

    while True:
        i = dType.GetInfraredSensor(api, 1)
        if i[0] == 1:
            break

    dType.SetEMotor(api, 0, 0, -10000, 1)                 # 停传送带
    socket_client.send("arrive2".encode('utf-8'))
```
